# ============================================================
# Linux Master Lab
# Amazon Linux 2023

MODULES="$LAB_ROOT/modules"

echo "=========================================="
echo " Linux Master Lab - Initializing"
echo "=========================================="

# ------------------------------------------------------------
# 1. Install only lightweight tools
# ------------------------------------------------------------

dnf install -y \
    tree \
    vim-enhanced \
    nano \
    tar \
    gzip \
    unzip \
    zip

# ------------------------------------------------------------
# 2. Create lab directories
# ------------------------------------------------------------

mkdir -p "$MODULES"
mkdir -p /usr/local/bin

# ------------------------------------------------------------
# 3. Create Linux Lab command
# ------------------------------------------------------------

cat > /usr/local/bin/linux-lab <<'LABCOMMAND'
#!/bin/bash

LAB_ROOT="/opt/linux-lab"
MODULES="$LAB_ROOT/modules"
FUNDAMENTALS="$MODULES/fundamentals"

GREEN='\033[0;32m'
YELLOW='\033[1;33m'
RED='\033[0;31m'
BLUE='\033[0;34m'
NC='\033[0m'

show_usage() {

    echo
    echo "=========================================="
    echo "        LINUX MASTER LAB"
    echo "=========================================="
    echo
    echo "Usage:"
    echo
    echo "  linux-lab list"
    echo "  linux-lab start fundamentals"
    echo "  linux-lab status"
    echo "  linux-lab check fundamentals"
    echo "  linux-lab clean fundamentals"
    echo
}

# ============================================================
# LIST MODULES
# ============================================================

list_modules() {

    echo
    echo "Available Modules"
    echo "-----------------"
    echo
    echo "  01  fundamentals"
    echo
}

# ============================================================
# CREATE FUNDAMENTALS MODULE
# ============================================================

        echo
        echo "Location:"
        echo "$FUNDAMENTALS"
        echo

        return
    fi

    echo
    echo -e "${BLUE}Starting Linux Fundamentals module...${NC}"
    echo

    # --------------------------------------------------------
    # Directory structure
    # --------------------------------------------------------

    mkdir -p "$FUNDAMENTALS/workspace/projects/webapp"
    mkdir -p "$FUNDAMENTALS/workspace/projects/api"
    mkdir -p "$FUNDAMENTALS/workspace/documents"
    mkdir -p "$FUNDAMENTALS/workspace/backups"
    mkdir -p "$FUNDAMENTALS/workspace/temporary"

    mkdir -p "$FUNDAMENTALS/data"

    # --------------------------------------------------------

    cat > "$FUNDAMENTALS/data/products.csv" <<EOF

    cat > "$FUNDAMENTALS/data/servers.txt" <<EOF

    # Sample application files
    # --------------------------------------------------------

print("Linux Master Lab API")

    cat > "$FUNDAMENTALS/workspace/documents/readme.txt" <<EOF
Linux Master Lab

Module: Fundamentals

Practice Linux commands using this environment.
EOF

    # --------------------------------------------------------
    # Exercise file
    # --------------------------------------------------------

    cat > "$FUNDAMENTALS/exercises/tasks.md" <<'EOF'

# ============================================================
# STATUS
# ============================================================

status() {

    echo
    echo "=========================================="
    echo "        LINUX MASTER LAB STATUS"
    echo "=========================================="
    echo

    if [ -d "$FUNDAMENTALS" ]; then

        echo -e "Fundamentals : ${GREEN}ACTIVE${NC}"

        echo
        echo "Location:"
        echo "$FUNDAMENTALS"

        echo
        echo "Files:"
        find "$FUNDAMENTALS" -type f | wc -l

        echo "Directories:"
        find "$FUNDAMENTALS" -type d | wc -l

    else

        echo "Fundamentals : NOT ACTIVE"
    fi

    echo
}

# ============================================================
# CHECK FUNDAMENTALS
# ============================================================

check_fundamentals() {

    if [ ! -d "$FUNDAMENTALS" ]; then

        echo
        echo -e "${RED}Fundamentals module is not active.${NC}"
        echo

        return 1
    fi

    echo
    echo "=========================================="
    echo "      FUNDAMENTALS PROGRESS CHECK"
    echo "=========================================="
    echo

    PASS=0
    TOTAL=8

    # 1
    if [ -d "$FUNDAMENTALS/workspace/projects/webapp/deployments" ]; then
        echo -e "${GREEN}[✓]${NC} deployments directory"
        ((PASS++))
    else
        echo "[ ] deployments directory"
    fi

    # 2
    if [ -d "$FUNDAMENTALS/workspace/projects/webapp/deployments/production" ]; then
        echo -e "${GREEN}[✓]${NC} production directory"
        ((PASS++))
    else
        echo "[ ] production directory"
    fi

    # 3
    if [ -f "$FUNDAMENTALS/workspace/projects/webapp/deployments/production/release.txt" ]; then
        echo -e "${GREEN}[✓]${NC} release.txt created/renamed"
        ((PASS++))
    else
        echo "[ ] release.txt"
    fi

    # 4
    if [ -f "$FUNDAMENTALS/workspace/backups/v1.txt" ]; then
        echo -e "${GREEN}[✓]${NC} v1.txt backup"
        ((PASS++))
    else
        echo "[ ] v1.txt backup"
    fi

    # 5
    if [ -f "$FUNDAMENTALS/workspace/temporary/v3.txt" ]; then
        echo -e "${GREEN}[✓]${NC} v3.txt moved"
        ((PASS++))
    else
        echo "[ ] v3.txt movement"
    fi

    # 6
    if [ -L "$FUNDAMENTALS/workspace/projects/webapp/release-link" ]; then
        echo -e "${GREEN}[✓]${NC} symbolic link"
        ((PASS++))
    else
        echo "[ ] symbolic link"
    fi

    # 7
    if [ -f "$FUNDAMENTALS/workspace/backups/webapp.tar" ] || \
       [ -f "$FUNDAMENTALS/workspace/backups/webapp.tar.gz" ]; then
        echo -e "${GREEN}[✓]${NC} webapp archive"
        ((PASS++))
    else
        echo "[ ] webapp archive"
    fi

    # 8
    if [ -d "$FUNDAMENTALS/workspace/extracted" ]; then
        echo -e "${GREEN}[✓]${NC} archive extraction directory"
        ((PASS++))
    else
        echo "[ ] archive extraction"
    fi

    echo
    echo "Progress: $PASS/$TOTAL"
    echo

    if [ "$PASS" -eq "$TOTAL" ]; then
        echo -e "${GREEN}MODULE CHECK: PASSED${NC}"
        echo
        echo "You can move to the next module."
    else
        echo -e "${YELLOW}MODULE CHECK: INCOMPLETE${NC}"
        echo
        echo "Continue practicing the failed tasks."
    fi

    echo
}

# ============================================================
# CLEAN FUNDAMENTALS
# ============================================================

clean_fundamentals() {

    if [ ! -d "$FUNDAMENTALS" ]; then

        echo
        echo "Fundamentals module is not active."
        echo

        return
    fi

    echo
    echo -e "${YELLOW}Cleaning Fundamentals module...${NC}"
    echo

    rm -rf "$FUNDAMENTALS"

    echo -e "${GREEN}[✓] Fundamentals module removed.${NC}"
    echo
    echo "The EC2 instance is ready for the next module."
    echo
}

# ============================================================
# COMMAND ROUTER
# ============================================================

case "$1" in

    list)
        list_modules
        ;;

    start)

        case "$2" in

            fundamentals)
                start_fundamentals
                ;;

            *)
                echo
                echo "Unknown module: $2"
                echo
                ;;

        esac

        ;;

    status)
        status
        ;;

    check)

        case "$2" in

            fundamentals)
                check_fundamentals
                ;;

            *)
                echo
                echo "Unknown module: $2"
                echo
                ;;

        esac

        ;;

    clean)

        case "$2" in

            fundamentals)
                clean_fundamentals
                ;;

            *)
                echo
                echo "Unknown module: $2"
                echo
                ;;

        esac

        ;;

    *)
        show_usage
        ;;

esac

LABCOMMAND

chmod +x /usr/local/bin/linux-lab

# ------------------------------------------------------------
# 4. Create initial README
# ------------------------------------------------------------

cat > "$LAB_ROOT/README.md" <<'EOF'
============================================================
LINUX MASTER LAB
============================================================

This EC2 instance contains the Linux Master Lab.

Available commands:

    linux-lab list

    linux-lab start fundamentals

    linux-lab status

    linux-lab check fundamentals

    linux-lab clean fundamentals


Learning model:

    START
      ↓
    PRACTICE
      ↓
    CHECK
      ↓
    CLEAN
      ↓
    NEXT MODULE

Only the active module creates practice resources.

This keeps the EC2 lightweight.
EOF

# ------------------------------------------------------------
# 5. Installation marker
# ------------------------------------------------------------

cat > "$LAB_ROOT/.installed" <<EOF
Linux Master Lab installed.
Installed: $(date)
OS: Amazon Linux 2023
EOF

echo "=========================================="
echo " Linux Master Lab READY"
echo "=========================================="
echo
echo "SSH into the instance and run:"
echo
echo "    linux-lab list"
echo
echo "Then:"
echo
echo "    linux-lab start fundamentals"

echo "=========================================="
